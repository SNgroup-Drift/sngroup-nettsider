export function Brand({ size = 20, product = 'CRM', crm, markOnly, wordmark = true, href, style }) {
  const only = markOnly || wordmark === false;
  const prod = crm === false ? null : product;
  const Tag = href ? 'a' : 'span';
  if (only) return h(Tag, { className: 'h-brand only', href, style: { fontSize: size, ...style }, role: 'img', 'aria-label': 'Hengsel' }, h(Mark));
  return h(Tag, { className: 'h-brand' + (prod ? ' has-product' : ''), href, style: { fontSize: size, ...style }, 'aria-label': prod ? 'Hengsel ' + prod : 'Hengsel' },
    h(Mark),
    h('span', { className: 'h-brand-word', 'aria-hidden': true }, 'engsel'),
    prod && h('svg', { viewBox: '0 0 30 26', className: 'h-brand-check', 'aria-hidden': true },
      h('path', { d: 'M3 14l8 8l16 -18', fill: 'none', stroke: 'var(--mark-check)', strokeWidth: 4.6, strokeLinecap: 'round', strokeLinejoin: 'round' })),
    prod && h('span', { className: 'h-brand-product', 'aria-hidden': true }, prod));
}
